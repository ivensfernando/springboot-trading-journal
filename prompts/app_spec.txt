<project_specification>
  <project_name>Trading Journal</project_name>

  <overview>
    A single-user cryptocurrency trading platform that tracks trades and performance, analyzes trading patterns, imports trade history automatically from exchanges (KuCoin, MEXC, Phemex), and executes trades directly through exchange APIs. This is a portfolio/example project designed to demonstrate proficiency in Java, Maven, and Spring Boot technologies for GitHub showcase.
  </overview>

    <frontend>

      <state_management>React Context or Redux Toolkit</state_management>
      <http_client>Axios</http_client>
      <charts>Recharts or Chart.js</charts>
    </frontend>
    <backend>
      <runtime>Java 17+</runtime>
      <framework>Spring Boot 3.x</framework>
      <build_tool>Maven</build_tool>

      <logging>SLF4J with Logback</logging>
      <async>CompletableFuture</async>

    </backend>
    <database>

      <orm>Spring Data JPA with Hibernate</orm>
      <migrations>Flyway</migrations>

  <database_schema>
    <tables>
      <users>
        - id (UUID, primary key)
        - username (unique, not null)
        - password_hash (not null)
        - email (unique)
        - created_at (timestamp)
        - updated_at (timestamp)
        - last_login (timestamp)
      </users>

      <exchange_connections>
        - id (UUID, primary key)
        - user_id (foreign key to users)
        - exchange_name (enum: KUCOIN, MEXC, PHEMEX)
        - api_key_encrypted (not null)
        - api_secret_encrypted (not null)
        - api_passphrase_encrypted (nullable, for KuCoin)
        - is_active (boolean)
        - last_sync (timestamp)
        - created_at (timestamp)
        - updated_at (timestamp)
      </exchange_connections>

      <trades>
        - id (UUID, primary key)
        - user_id (foreign key to users)
        - exchange_connection_id (foreign key)
        - external_trade_id (from exchange)
        - symbol (trading pair, e.g., BTC/USDT)
        - side (enum: BUY, SELL)
        - entry_price (decimal)
        - exit_price (decimal, nullable)
        - quantity (decimal)
        - fees (decimal)
        - fee_currency (string)
        - realized_pnl (decimal, nullable)
        - status (enum: OPEN, CLOSED, CANCELLED)
        - notes (text, nullable)
        - strategy (string, nullable)
        - is_imported (boolean)
        - trade_date (timestamp)
        - created_at (timestamp)
        - updated_at (timestamp)
      </trades>

      <trade_tags>
        - id (UUID, primary key)
        - trade_id (foreign key to trades)
        - tag_name (string)
      </trade_tags>

      <positions>
        - id (UUID, primary key)
        - user_id (foreign key to users)
        - exchange_connection_id (foreign key)
        - symbol (trading pair)
        - side (enum: LONG, SHORT)
        - entry_price (decimal)
        - quantity (decimal)
        - current_price (decimal)
        - unrealized_pnl (decimal)
        - unrealized_pnl_percent (decimal)
        - is_external (boolean, true if opened outside app)
        - opened_at (timestamp)
        - last_updated (timestamp)
      </positions>

      <orders>
        - id (UUID, primary key)
        - user_id (foreign key to users)
        - exchange_connection_id (foreign key)
        - external_order_id (from exchange)
        - symbol (trading pair)
        - side (enum: BUY, SELL)
        - order_type (enum: MARKET, LIMIT, STOP_LOSS)
        - price (decimal, nullable for market)
        - quantity (decimal)
        - filled_quantity (decimal)
        - status (enum: PENDING, FILLED, PARTIALLY_FILLED, CANCELLED, REJECTED)
        - created_at (timestamp)
        - updated_at (timestamp)
      </orders>

      <user_settings>
        - id (UUID, primary key)
        - user_id (foreign key to users, unique)
        - default_exchange (enum, nullable)
        - default_order_size (decimal, nullable)
        - theme (enum: LIGHT, DARK)
        - auto_refresh_interval (integer, seconds)
        - timezone (string)
        - created_at (timestamp)
        - updated_at (timestamp)
      </user_settings>

      <strategies>
        - id (UUID, primary key)
        - user_id (foreign key to users)
        - name (string, unique per user)
        - description (text, nullable)
        - created_at (timestamp)
      </strategies>
    </tables>
  </database_schema>
